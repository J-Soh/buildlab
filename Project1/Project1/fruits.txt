#include "cJSON.c"
#include "cprocessing.h"
#include <math.h>
#include <stdbool.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define SCREEN_WIDTH 1280.0f
#define SCREEN_HEIGHT 720.0f
#define TILE_SIZE 48.0f // Scaled tile size for rendering

// Structure to store parsed Fruit Object positions
typedef struct {
    float x;
    float y;
} GameObjectPos;

// Holds texture, frame count, and positions for a specific fruit type
typedef struct {
    CP_Image image;
    int totalFrames;
    GameObjectPos* positions;
    int count;
} FruitGroup;

// Enum for all fruit types
typedef enum {
    FRUIT_ORANGE,
    FRUIT_CHERRIES,
    FRUIT_KIWI,
    FRUIT_MELON,
    FRUIT_STRAWBERRY,
    FRUIT_COUNT
} FruitType;

static FruitGroup fruits[FRUIT_COUNT];

// State Machine for managing screens
typedef enum {
    STATE_MENU,
    STATE_GAME
} GameState;

// Global Variables
static GameState currentState = STATE_MENU;
static bool showSettings = false; // Toggle to view instructions overlay
static CP_Image bgImage = NULL;
static CP_Font gameFont = NULL;

// Button Image Handles
static CP_Image playBtnImage = NULL;
static CP_Image volBtnImage = NULL;
static CP_Image settingsBtnImage = NULL;

// Transition Handles & Variables
static CP_Image transitionImage = NULL;
static bool isTransitioning = false;
static float transitionTimer = 0.0f;
static const float TRANSITION_DURATION = 0.7f; // Transition duration in seconds

// Terrain Tilesheet Handle & Dimensions
static CP_Image terrainSheet = NULL;
static float sheetW = 384.0f; // Width of Terrain (16x16).png
static float sheetH = 192.0f; // Height of Terrain (16x16).png

// Tilemap Camera (world-pixel position of the viewport's top-left corner)
static float camX = 0.0f;
static float camY = 0.0f;
static const float CAM_SPEED = 500.0f; // world px per second

// Map JSON Storage
static int mapWidth = 0;
static int mapHeight = 0;
static int tileWidth = 16;
static int tileHeight = 16;
static int* mapData = NULL;


// -----------------------------------------------------------------------------
// Helper Functions ("Methods")
// -----------------------------------------------------------------------------

// Checks if mouse position is inside an image bounding box
static bool is_mouse_over(float x, float y, float width, float height) {
    float mx = CP_Input_GetMouseX();
    float my = CP_Input_GetMouseY();
    return (mx >= x && mx <= x + width && my >= y && my <= y + height);
}

// Tiles Pink.png across the screen to fill 1280x720
static void draw_background(void) {
    if (!bgImage) return;

    float bgWidth = (float)CP_Image_GetWidth(bgImage);
    float bgHeight = (float)CP_Image_GetHeight(bgImage);

    for (float x = 0.0f; x < SCREEN_WIDTH; x += bgWidth) {
        for (float y = 0.0f; y < SCREEN_HEIGHT; y += bgHeight) {
            CP_Image_Draw(bgImage, x, y, bgWidth, bgHeight, 255);
        }
    }
}

// Map Tiled layer name to FruitType enum
static FruitType get_fruit_type_from_name(const char* name) {
    if (!name) return FRUIT_COUNT;
    if (strstr(name, "Orange") || strstr(name, "orange")) return FRUIT_ORANGE;
    if (strstr(name, "Cherries") || strstr(name, "cherries")) return FRUIT_CHERRIES;
    if (strstr(name, "Kiwi") || strstr(name, "kiwi")) return FRUIT_KIWI;
    if (strstr(name, "Melon") || strstr(name, "melon")) return FRUIT_MELON;
    if (strstr(name, "Strawberry") || strstr(name, "strawberry")) return FRUIT_STRAWBERRY;
    return FRUIT_COUNT;
}

// Helper to extract objects from an objectgroup layer into the appropriate FruitGroup
static void parse_object_group(cJSON* layer) {
    cJSON* layerName = cJSON_GetObjectItemCaseSensitive(layer, "name");
    FruitType type = FRUIT_ORANGE; // Default fallback

    if (cJSON_IsString(layerName) && layerName->valuestring) {
        FruitType detected = get_fruit_type_from_name(layerName->valuestring);
        if (detected != FRUIT_COUNT) type = detected;
    }

    cJSON* objects = cJSON_GetObjectItemCaseSensitive(layer, "objects");
    if (!cJSON_IsArray(objects)) return;

    int objCount = cJSON_GetArraySize(objects);
    if (objCount == 0) return;

    // Allocate memory for this fruit type
    if (fruits[type].positions) free(fruits[type].positions);
    fruits[type].positions = (GameObjectPos*)malloc(sizeof(GameObjectPos) * objCount);
    fruits[type].count = 0;

    for (int i = 0; i < objCount; i++) {
        cJSON* obj = cJSON_GetArrayItem(objects, i);
        if (!obj) continue;

        cJSON* x = cJSON_GetObjectItemCaseSensitive(obj, "x");
        cJSON* y = cJSON_GetObjectItemCaseSensitive(obj, "y");

        if (cJSON_IsNumber(x) && cJSON_IsNumber(y)) {
            fruits[type].positions[fruits[type].count].x = (float)x->valuedouble;
            fruits[type].positions[fruits[type].count].y = (float)y->valuedouble;
            fruits[type].count++;
        }
    }
}

// Recursively processes layers to find tilelayers and objectgroups (even inside group layers)
static void process_layers(cJSON* layers) {
    if (!cJSON_IsArray(layers)) return;

    int layerCount = cJSON_GetArraySize(layers);
    for (int l = 0; l < layerCount; l++) {
        cJSON* layer = cJSON_GetArrayItem(layers, l);
        if (!layer) continue;

        cJSON* type = cJSON_GetObjectItemCaseSensitive(layer, "type");
        if (!type || !type->valuestring) continue;

        // Tile layer parsing
        if (strcmp(type->valuestring, "tilelayer") == 0) {
            cJSON* data = cJSON_GetObjectItemCaseSensitive(layer, "data");
            if (cJSON_IsArray(data)) {
                int dataSize = cJSON_GetArraySize(data);
                if (mapData) free(mapData);
                mapData = (int*)malloc(sizeof(int) * dataSize);

                for (int i = 0; i < dataSize; i++) {
                    cJSON* tile = cJSON_GetArrayItem(data, i);
                    if (cJSON_IsNumber(tile)) {
                        mapData[i] = tile->valueint;
                    }
                }
            }
        }
        // Object group layer parsing
        else if (strcmp(type->valuestring, "objectgroup") == 0) {
            parse_object_group(layer);
        }
        // Group layer parsing (recurse into nested layers)
        else if (strcmp(type->valuestring, "group") == 0) {
            cJSON* subLayers = cJSON_GetObjectItemCaseSensitive(layer, "layers");
            if (subLayers) {
                process_layers(subLayers);
            }
        }
    }
}

// Loads and parses map.json exported from Tiled using cJSON
static void load_map_json(const char* filepath) {
    FILE* file = fopen(filepath, "rb");
    if (!file) {
        printf("FAILED to open map: %s, trying fallbacks...\n", filepath);
        const char* fallbacks[] = {
            "../Assets/Terrain/map.json",
            "../../Assets/Terrain/map.json",
            "Terrain/map.json",
            "map.json"
        };
        for (int i = 0; i < 4 && !file; i++) {
            file = fopen(fallbacks[i], "rb");
            if (file) printf("Opened map via fallback: %s\n", fallbacks[i]);
        }
        if (!file) return;
    }

    fseek(file, 0, SEEK_END);
    long length = ftell(file);
    fseek(file, 0, SEEK_SET);

    char* buffer = (char*)malloc((size_t)length + 1);
    if (!buffer) {
        fclose(file);
        return;
    }

    fread(buffer, 1, length, file);
    buffer[length] = '\0';
    fclose(file);

    cJSON* json = cJSON_Parse(buffer);
    free(buffer);
    if (!json) return;

    cJSON* width = cJSON_GetObjectItemCaseSensitive(json, "width");
    cJSON* height = cJSON_GetObjectItemCaseSensitive(json, "height");
    cJSON* tilew = cJSON_GetObjectItemCaseSensitive(json, "tilewidth");
    cJSON* tileh = cJSON_GetObjectItemCaseSensitive(json, "tileheight");

    if (cJSON_IsNumber(width)) mapWidth = width->valueint;
    if (cJSON_IsNumber(height)) mapHeight = height->valueint;
    if (cJSON_IsNumber(tilew)) tileWidth = tilew->valueint;
    if (cJSON_IsNumber(tileh)) tileHeight = tileh->valueint;

    cJSON* layers = cJSON_GetObjectItemCaseSensitive(json, "layers");
    if (layers) {
        process_layers(layers);
    }

    cJSON_Delete(json);

    printf("Map loaded: %dx%d, tile %dx%d\n", mapWidth, mapHeight, tileWidth, tileHeight);
    for (int i = 0; i < FRUIT_COUNT; i++) {
        printf("Fruit %d items parsed: %d\n", i, fruits[i].count);
    }
}

// Renders the tilemap parsed from map.json using Terrain (16x16).png
static void render_map_json(void) {
    if (!mapData || !terrainSheet) return;

    CP_Settings_ImageMode(CP_POSITION_CORNER);
    int tilesPerRowInSheet = (int)(sheetW / tileWidth);
    if (tilesPerRowInSheet < 1) tilesPerRowInSheet = 1;

    for (int y = 0; y < mapHeight; y++) {
        for (int x = 0; x < mapWidth; x++) {
            int tileIndex = y * mapWidth + x;
            int tileGid = mapData[tileIndex];

            if (tileGid <= 0) continue;

            float screenX = x * TILE_SIZE - camX;
            float screenY = y * TILE_SIZE - camY;

            if (screenX < -TILE_SIZE || screenX > SCREEN_WIDTH ||
                screenY < -TILE_SIZE || screenY > SCREEN_HEIGHT) continue;

            int tileId = tileGid - 1;
            float cropX = (float)((tileId % tilesPerRowInSheet) * tileWidth);
            float cropY = (float)((tileId / tilesPerRowInSheet) * tileHeight);

            CP_Image_DrawSubImage(terrainSheet, screenX, screenY, TILE_SIZE, TILE_SIZE, cropX, cropY, cropX + (float)tileWidth, cropY + (float)tileHeight, 255);
        }
    }
}

// Loads a fruit image asset and calculates frame count automatically
void load_fruit_type(FruitType type, const char* imagePath) {
    fruits[type].image = CP_Image_Load(imagePath);
    fruits[type].positions = NULL;
    fruits[type].count = 0;

    if (fruits[type].image) {
        float height = (float)CP_Image_GetHeight(fruits[type].image); // 32px
        float width = (float)CP_Image_GetWidth(fruits[type].image);   // Total width
        fruits[type].totalFrames = (int)(width / height);
    }
}

// Renders all animated fruit objects in world coordinates
static void render_objects(void) {
    CP_Settings_ImageMode(CP_POSITION_CORNER);
    float scaleRatio = TILE_SIZE / (float)tileWidth;

    for (int f = 0; f < FRUIT_COUNT; f++) {
        FruitGroup* fruit = &fruits[f];
        if (!fruit->image || fruit->count == 0 || fruit->totalFrames == 0) continue;

        // Frame dimensions (32x32px per frame)
        float frameHeight = (float)CP_Image_GetHeight(fruit->image);
        float frameWidth = frameHeight;

        // Animate based on total frames for this specific fruit
        int currentFrame = (int)(CP_System_GetSeconds() * 12.0f) % fruit->totalFrames;
        float cropX1 = currentFrame * frameWidth;
        float cropX2 = cropX1 + frameWidth;

        for (int i = 0; i < fruit->count; i++) {
            float worldX = fruit->positions[i].x * scaleRatio;
            float worldY = fruit->positions[i].y * scaleRatio;

            float screenX = worldX - camX;
            float screenY = worldY - camY;

            // Off-screen culling
            if (screenX < -TILE_SIZE || screenX > SCREEN_WIDTH ||
                screenY < -TILE_SIZE || screenY > SCREEN_HEIGHT) continue;

            CP_Image_DrawSubImage(
                fruit->image,
                screenX, screenY,
                TILE_SIZE, TILE_SIZE,
                cropX1, 0.0f,
                cropX2, frameHeight,
                255
            );
        }
    }
}

// Renders the scaling transition over screen
static void draw_transition(void) {
    if (!isTransitioning || !transitionImage) return;

    transitionTimer += CP_System_GetDt();
    float progress = transitionTimer / TRANSITION_DURATION;

    if (progress <= 0.5f) {
        float expandProgress = progress * 2.0f;
        float maxScale = 2500.0f;
        float currentSize = expandProgress * maxScale;

        CP_Settings_ImageMode(CP_POSITION_CENTER);
        CP_Image_Draw(transitionImage, SCREEN_WIDTH * 0.5f, SCREEN_HEIGHT * 0.5f, currentSize, currentSize, 255);
        CP_Settings_ImageMode(CP_POSITION_CORNER);
    }
    else if (progress < 1.0f) {
        currentState = STATE_GAME;

        float shrinkProgress = 1.0f - ((progress - 0.5f) * 2.0f);
        float maxScale = 2500.0f;
        float currentSize = shrinkProgress * maxScale;

        CP_Settings_ImageMode(CP_POSITION_CENTER);
        CP_Image_Draw(transitionImage, SCREEN_WIDTH * 0.5f, SCREEN_HEIGHT * 0.5f, currentSize, currentSize, 255);
        CP_Settings_ImageMode(CP_POSITION_CORNER);
    }
    else {
        isTransitioning = false;
        transitionTimer = 0.0f;
    }
}

// Draws the Title Screen UI
static void draw_menu_ui(void) {
    CP_Settings_RectMode(CP_POSITION_CENTER);

    float centerX = SCREEN_WIDTH * 0.5f;
    float centerY = SCREEN_HEIGHT * 0.5f;
    float panelW = 800.0f;
    float panelH = 500.0f;

    CP_Settings_Fill(CP_Color_Create(40, 20, 50, 220));
    CP_Graphics_DrawRect(centerX, centerY, panelW + 16.0f, panelH + 16.0f);

    CP_Settings_Fill(CP_Color_Create(245, 230, 210, 240));
    CP_Graphics_DrawRect(centerX, centerY, panelW, panelH);

    CP_Settings_NoFill();
    CP_Settings_Stroke(CP_Color_Create(215, 140, 40, 255));
    CP_Settings_StrokeWeight(6.0f);
    CP_Graphics_DrawRect(centerX, centerY, panelW - 24.0f, panelH - 24.0f);
    CP_Settings_NoStroke();

    CP_Settings_RectMode(CP_POSITION_CORNER);

    if (gameFont) {
        CP_Font_Set(gameFont);
    }

    CP_Settings_TextAlignment(CP_TEXT_ALIGN_H_CENTER, CP_TEXT_ALIGN_V_MIDDLE);

    if (showSettings) {
        CP_Settings_Fill(CP_Color_Create(180, 50, 40, 255));
        CP_Settings_TextSize(54.0f);
        CP_Font_DrawText("CONTROLS & SETTINGS", centerX, 220.0f);

        CP_Settings_Fill(CP_Color_Create(40, 40, 40, 255));
        CP_Settings_TextSize(32.0f);
        CP_Font_DrawText("WASD  -  Move Left / Right", centerX, 300.0f);
        CP_Font_DrawText("SPACE  -  Jump", centerX, 350.0f);

        CP_Settings_Fill(CP_Color_Create(180, 50, 40, 255));
        CP_Settings_TextSize(28.0f);
        CP_Font_DrawText("Click anywhere on card to go BACK", centerX, 440.0f);
    }
    else {
        CP_Settings_Fill(CP_Color_Create(180, 50, 40, 255));
        CP_Settings_TextSize(72.0f);
        CP_Font_DrawText("PIXEL ADVENTURE", centerX, 220.0f);

        CP_Settings_Fill(CP_Color_Create(80, 80, 80, 255));
        CP_Settings_TextSize(28.0f);
        CP_Font_DrawText("Press [ SPACE ] or click PLAY button to Start", centerX, 290.0f);

        float btnSize = 64.0f;
        float btnY = 400.0f;

        if (playBtnImage) {
            CP_Image_Draw(playBtnImage, centerX - (btnSize * 0.5f), btnY, btnSize, btnSize, 255);
        }

        if (volBtnImage) {
            CP_Image_Draw(volBtnImage, centerX - (btnSize * 0.5f) - 100.0f, btnY, btnSize, btnSize, 255);
        }

        if (settingsBtnImage) {
            CP_Image_Draw(settingsBtnImage, centerX - (btnSize * 0.5f) + 100.0f, btnY, btnSize, btnSize, 255);
        }
    }
}

// Handles input during the menu screen
static void handle_menu_input(void) {
    if (isTransitioning) return;

    if (CP_Input_KeyTriggered(KEY_SPACE)) {
        isTransitioning = true;
        transitionTimer = 0.0f;
    }

    if (CP_Input_MouseClicked()) {
        float centerX = SCREEN_WIDTH * 0.5f;

        if (showSettings) {
            float panelW = 800.0f;
            float panelH = 500.0f;
            if (is_mouse_over(centerX - (panelW * 0.5f), (SCREEN_HEIGHT - panelH) * 0.5f, panelW, panelH)) {
                showSettings = false;
            }
        }
        else {
            float btnSize = 64.0f;
            float btnY = 400.0f;

            float playX = centerX - (btnSize * 0.5f);
            float settingsX = playX + 100.0f;

            if (is_mouse_over(playX, btnY, btnSize, btnSize)) {
                isTransitioning = true;
                transitionTimer = 0.0f;
            }

            if (is_mouse_over(settingsX, btnY, btnSize, btnSize)) {
                showSettings = true;
            }
        }
    }
}

// -----------------------------------------------------------------------------
// Lifecycle Functions
// -----------------------------------------------------------------------------

void game_init(void) {
    CP_System_SetWindowSize((int)SCREEN_WIDTH, (int)SCREEN_HEIGHT);
    CP_System_SetWindowTitle("Pixel Adventure");

    CP_Settings_ImageMode(CP_POSITION_CORNER);
    CP_Settings_ImageFilterMode(CP_IMAGE_FILTER_NEAREST);

    bgImage = CP_Image_Load("Assets/Background/Brown.png");
    gameFont = CP_Font_Load("Assets/ThaleahFat.ttf");

    playBtnImage = CP_Image_Load("Assets/Menu/Buttons/Play.png");
    volBtnImage = CP_Image_Load("Assets/Menu/Buttons/Volume.png");
    settingsBtnImage = CP_Image_Load("Assets/Menu/Buttons/Settings.png");

    transitionImage = CP_Image_Load("Assets/Other/Transition.png");

    terrainSheet = CP_Image_Load("Assets/Terrain/Terrain (16x16).png");
    if (terrainSheet) {
        sheetW = (float)CP_Image_GetWidth(terrainSheet);
        sheetH = (float)CP_Image_GetHeight(terrainSheet);
    }

    // Load All Fruit Asset Images
    load_fruit_type(FRUIT_ORANGE, "Assets/Items/Fruits/Orange.png");
    load_fruit_type(FRUIT_CHERRIES, "Assets/Items/Fruits/Cherries.png");
    load_fruit_type(FRUIT_KIWI, "Assets/Items/Fruits/Kiwi.png");
    load_fruit_type(FRUIT_MELON, "Assets/Items/Fruits/Melon.png");
    load_fruit_type(FRUIT_STRAWBERRY, "Assets/Items/Fruits/Strawberry.png");

    load_map_json("Assets/Terrain/map.json");

    camX = 1200.0f;
    camY = 1700.0f;
}

void game_update(void) {
    if (currentState == STATE_MENU) {
        CP_Graphics_ClearBackground(CP_Color_Create(0, 0, 0, 255));
        draw_background();

        draw_menu_ui();
        handle_menu_input();
        draw_transition();
    }
    else if (currentState == STATE_GAME) {
        CP_Graphics_ClearBackground(CP_Color_Create(33, 31, 48, 255));

        float dt = CP_System_GetDt();
        if (CP_Input_KeyDown(KEY_W)) camY -= CAM_SPEED * dt;
        if (CP_Input_KeyDown(KEY_S)) camY += CAM_SPEED * dt;
        if (CP_Input_KeyDown(KEY_A)) camX -= CAM_SPEED * dt;
        if (CP_Input_KeyDown(KEY_D)) camX += CAM_SPEED * dt;

        camX = roundf(camX);
        camY = roundf(camY);

        draw_background();

        render_map_json();
        render_objects(); // Draw all registered fruit items over the map

        draw_transition();
    }
}

void game_exit(void) {
    if (bgImage) CP_Image_Free(bgImage);
    if (gameFont) CP_Font_Free(gameFont);

    if (playBtnImage) CP_Image_Free(playBtnImage);
    if (volBtnImage) CP_Image_Free(volBtnImage);
    if (settingsBtnImage) CP_Image_Free(settingsBtnImage);
    if (transitionImage) CP_Image_Free(transitionImage);
    if (terrainSheet) CP_Image_Free(terrainSheet);

    // Free memory for all fruit groups
    for (int i = 0; i < FRUIT_COUNT; i++) {
        if (fruits[i].image) CP_Image_Free(fruits[i].image);
        if (fruits[i].positions) free(fruits[i].positions);
    }

    if (mapData) free(mapData);
}

int main(void) {
    CP_Engine_SetNextGameState(game_init, game_update, game_exit);
    CP_Engine_Run(0);
    return 0;
}
